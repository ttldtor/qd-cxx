cmake_minimum_required(VERSION 3.25)
project(qdcxx_event_values_consumer LANGUAGES CXX)
option(ORG_TTLDTOR_QDCXX_CONSUMER_FEED "Consume only the installed feed target" OFF)
if(ORG_TTLDTOR_QDCXX_CONSUMER_FEED)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
else()
    find_package(qdcxx CONFIG REQUIRED)
endif()
add_executable(qdcxx_event_values_consumer main.cpp)
if(WIN32)
    target_sources(qdcxx_event_values_consumer PRIVATE windows_headers.cpp)
endif()
if(ORG_TTLDTOR_QDCXX_CONSUMER_FEED)
    target_link_libraries(qdcxx_event_values_consumer PRIVATE qdcxx::feed)
else()
    target_link_libraries(qdcxx_event_values_consumer PRIVATE qdcxx_event_values)
endif()
foreach(header IN ITEMS event_value market_enums event_flags time_helpers short_string)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${header}.cpp" "#include <qd/events/${header}.hpp>\n")
    target_sources(qdcxx_event_values_consumer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${header}.cpp")
endforeach()
set_target_properties(qdcxx_event_values_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME event_values_consumer COMMAND qdcxx_event_values_consumer)
set_tests_properties(event_values_consumer PROPERTIES TIMEOUT 20)
