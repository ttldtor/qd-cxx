cmake_minimum_required(VERSION 3.25)
project(qdcxx_schedule_query_consumer LANGUAGES CXX)
find_package(qdcxx CONFIG REQUIRED COMPONENTS schedule time_zone_data_backend)
if(NOT IS_ABSOLUTE "${ORG_TTLDTOR_QDCXX_TIME_ZONE_DATA_ROOT}")
    message(FATAL_ERROR "Set the explicit installed versioned timezone data directory")
endif()
add_executable(qdcxx_schedule_query_consumer main.cpp header_alone.cpp time_header_alone.cpp)
target_link_libraries(qdcxx_schedule_query_consumer PRIVATE qdcxx_schedule)
set_target_properties(qdcxx_schedule_query_consumer PROPERTIES CXX_EXTENSIONS OFF)
if(ORG_TTLDTOR_QDCXX_SCHEDULE_SAMPLE_SOURCE)
    add_executable(qdcxx_schedule_installed_example "${ORG_TTLDTOR_QDCXX_SCHEDULE_SAMPLE_SOURCE}")
    target_link_libraries(qdcxx_schedule_installed_example PRIVATE qdcxx_schedule)
    target_compile_options(qdcxx_schedule_installed_example PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
    set_target_properties(qdcxx_schedule_installed_example PROPERTIES CXX_EXTENSIONS OFF)
endif()
enable_testing()
add_test(NAME schedule_installed_consumer COMMAND qdcxx_schedule_query_consumer "${ORG_TTLDTOR_QDCXX_TIME_ZONE_DATA_ROOT}")
set_tests_properties(schedule_installed_consumer PROPERTIES TIMEOUT 120)
if(TARGET qdcxx_schedule_installed_example)
    add_test(NAME schedule_installed_example COMMAND qdcxx_schedule_installed_example --run --data "${ORG_TTLDTOR_QDCXX_TIME_ZONE_DATA_ROOT}")
    set_tests_properties(schedule_installed_example PROPERTIES TIMEOUT 120)
endif()
