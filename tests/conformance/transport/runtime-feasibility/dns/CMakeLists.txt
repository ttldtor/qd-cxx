cmake_minimum_required(VERSION 3.25)
project(qdcxx LANGUAGES NONE)

option(ORG_TTLDTOR_QDCXX_BUILD_DNS_FEASIBILITY "Build the qualified DNS feasibility tests" OFF)
if(NOT ORG_TTLDTOR_QDCXX_BUILD_DNS_FEASIBILITY)
  return()
endif()

enable_language(C CXX)
set(CARES_SOURCE_DIR "" CACHE PATH "Explicit source tree with the pinned qdcxx1 patch applied")
option(ORG_TTLDTOR_QDCXX_DNS_SANITIZERS "Instrument the C SDK and C++ probe with ASan and UBSan" OFF)
if(NOT EXISTS "${CARES_SOURCE_DIR}/CMakeLists.txt" OR NOT EXISTS "${CARES_SOURCE_DIR}/include/ares_version.h")
  message(FATAL_ERROR "CARES_SOURCE_DIR must name the explicitly patched c-ares source tree")
endif()
file(READ "${CARES_SOURCE_DIR}/include/ares_version.h" caresVersionHeader)
string(FIND "${caresVersionHeader}" "#define ARES_QDCXX_PROFILE 1" profileMarker)
string(FIND "${caresVersionHeader}" "\"1.34.9+qdcxx1\"" versionMarker)
if(profileMarker EQUAL -1 OR versionMarker EQUAL -1)
  message(FATAL_ERROR "This test requires the qualified c-ares 1.34.9+qdcxx1 variant")
endif()
if(ORG_TTLDTOR_QDCXX_DNS_SANITIZERS)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "The sanitizer profile requires GCC or Clang")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

set(CARES_STATIC ON CACHE BOOL "" FORCE)
set(CARES_SHARED OFF CACHE BOOL "" FORCE)
set(CARES_INSTALL OFF CACHE BOOL "" FORCE)
set(CARES_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(CARES_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CARES_THREADS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CARES_SOURCE_DIR}" cares)
add_executable(qdcxx_dns_qualified_probe qualified_dns_probe.cpp)
target_compile_features(qdcxx_dns_qualified_probe PRIVATE cxx_std_23)
target_link_libraries(qdcxx_dns_qualified_probe PRIVATE c-ares::cares)
if(WIN32)
  target_link_libraries(qdcxx_dns_qualified_probe PRIVATE ws2_32)
endif()
enable_testing()
add_test(NAME qdcxx_dns_qualified_udp_tcp COMMAND qdcxx_dns_qualified_probe)
set_tests_properties(qdcxx_dns_qualified_udp_tcp PROPERTIES TIMEOUT 20)
