cmake_minimum_required(VERSION 3.25)
project(qdcxx LANGUAGES NONE)
option(ORG_TTLDTOR_QDCXX_TLS_PROBE "Build the isolated runtime feasibility probe" OFF)
if(NOT ORG_TTLDTOR_QDCXX_TLS_PROBE)
    return()
endif()
enable_language(CXX)
if(NOT DEFINED ORG_TTLDTOR_QDCXX_ASIO_INCLUDE)
    message(FATAL_ERROR "An explicit pinned standalone Asio include directory is required")
endif()
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS SSL Crypto)
find_package(Threads REQUIRED)
add_executable(qdcxx_tls_probe probe.cpp)
target_compile_features(qdcxx_tls_probe PRIVATE cxx_std_23)
target_include_directories(qdcxx_tls_probe SYSTEM PRIVATE "${ORG_TTLDTOR_QDCXX_ASIO_INCLUDE}")
target_compile_definitions(qdcxx_tls_probe PRIVATE ASIO_STANDALONE ASIO_NO_DEPRECATED)
target_link_libraries(qdcxx_tls_probe PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
    target_compile_definitions(qdcxx_tls_probe PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(qdcxx_tls_probe PRIVATE ws2_32 crypt32)
endif()
if(ORG_TTLDTOR_QDCXX_SANITIZE AND NOT MSVC)
    target_compile_options(qdcxx_tls_probe PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(qdcxx_tls_probe PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME qdcxx_tls_probe_help COMMAND "${CMAKE_COMMAND}"
    "-DPROBE=$<TARGET_FILE:qdcxx_tls_probe>" -P "${CMAKE_CURRENT_SOURCE_DIR}/ValidateHelp.cmake")
set_tests_properties(qdcxx_tls_probe_help PROPERTIES TIMEOUT 30)
if(DEFINED ORG_TTLDTOR_QDCXX_TLS_FIXTURES)
    add_test(NAME qdcxx_tls_probe_loopback COMMAND qdcxx_tls_probe --run
        --fixtures "${ORG_TTLDTOR_QDCXX_TLS_FIXTURES}")
    set_tests_properties(qdcxx_tls_probe_loopback PROPERTIES TIMEOUT 30)
endif()
