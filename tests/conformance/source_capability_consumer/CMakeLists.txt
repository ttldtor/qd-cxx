cmake_minimum_required(VERSION 3.25)
project(qdcxx_source_capability_consumer LANGUAGES CXX)
option(ORG_TTLDTOR_QDCXX_CONSUMER_USE_FEED "Consume only the installed Feed target" OFF)
if(ORG_TTLDTOR_QDCXX_CONSUMER_USE_FEED)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
else()
    find_package(qdcxx CONFIG REQUIRED)
endif()
add_executable(qdcxx_source_capability_consumer main.cpp order_header.cpp selection_header.cpp)
if(ORG_TTLDTOR_QDCXX_CONSUMER_USE_FEED)
    target_link_libraries(qdcxx_source_capability_consumer PRIVATE qdcxx::feed)
else()
    target_link_libraries(qdcxx_source_capability_consumer PRIVATE qdcxx_event_sources)
endif()
set_target_properties(qdcxx_source_capability_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME source_capability_consumer COMMAND qdcxx_source_capability_consumer)
set_tests_properties(source_capability_consumer PROPERTIES TIMEOUT 10)
