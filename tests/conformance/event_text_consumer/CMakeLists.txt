# Internal caller: installed/moved package checks. Inputs: explicit CMAKE_PREFIX_PATH and optional
# CONSUMER_FEED/CONSUMER_ZONE_BACKEND flags. Outputs: consumer/sample executables and standalone header objects.
# Configuration/build errors fail normally; CTest returns nonzero for API/help/data-root regressions.
cmake_minimum_required(VERSION 3.25)
project(qdcxx_event_text_consumer LANGUAGES CXX)
option(ORG_TTLDTOR_QDCXX_CONSUMER_FEED "Also consume the installed compatible feed target" OFF)
if(ORG_TTLDTOR_QDCXX_CONSUMER_FEED)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
else()
    find_package(qdcxx CONFIG REQUIRED)
endif()
if(ORG_TTLDTOR_QDCXX_CONSUMER_ZONE_BACKEND)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS time_zone_data_backend)
endif()
add_executable(qdcxx_event_text_consumer main.cpp)
target_link_libraries(qdcxx_event_text_consumer PRIVATE qdcxx_event_text)
if(ORG_TTLDTOR_QDCXX_CONSUMER_FEED)
    target_link_libraries(qdcxx_event_text_consumer PRIVATE qdcxx::feed)
endif()
get_target_property(qdcxxIncludeDirs qdcxx_core INTERFACE_INCLUDE_DIRECTORIES)
list(GET qdcxxIncludeDirs 0 qdcxxIncludeRoot)
if(NOT IS_ABSOLUTE "${qdcxxIncludeRoot}" OR NOT EXISTS "${qdcxxIncludeRoot}/qd/events/event_text.hpp")
    message(FATAL_ERROR "An installed qdcxx include root with event_text.hpp is required")
endif()
file(GLOB_RECURSE projectHeaders LIST_DIRECTORIES FALSE "${qdcxxIncludeRoot}/qd/*.hpp")
set(headerSources)
foreach(header IN LISTS projectHeaders)
    file(RELATIVE_PATH relative "${qdcxxIncludeRoot}" "${header}")
    string(MAKE_C_IDENTIFIER "${relative}" identity)
    set(unit "${CMAKE_CURRENT_BINARY_DIR}/header-${identity}.cpp")
    file(WRITE "${unit}" "#include <${relative}>\n")
    list(APPEND headerSources "${unit}")
endforeach()
if(WIN32)
    # Focused regressions require real active SDK min/max macros without changing public headers.
    foreach(relative IN ITEMS qd/events/event_text.hpp qd/core/serial_field_type.hpp)
        string(MAKE_C_IDENTIFIER "windows-first-${relative}" identity)
        set(unit "${CMAKE_CURRENT_BINARY_DIR}/header-${identity}.cpp")
        file(WRITE "${unit}" "#include <Windows.h>\n#if defined(NOMINMAX) || !defined(min) || !defined(max)\n#error Active Windows SDK min/max macros are required by this focused regression.\n#endif\n#include <${relative}>\n")
        list(APPEND headerSources "${unit}")
    endforeach()
endif()
list(LENGTH projectHeaders headerCount)
message(STATUS "Standalone installed public headers: ${headerCount}")
add_library(qdcxx_event_text_headers OBJECT ${headerSources})
target_link_libraries(qdcxx_event_text_headers PRIVATE qdcxx_event_text)
add_executable(qdcxx_event_text_installed_example ../../../examples/event_text/main.cpp)
target_link_libraries(qdcxx_event_text_installed_example PRIVATE qdcxx_event_text)
target_compile_options(qdcxx_event_text_installed_example PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
set_target_properties(qdcxx_event_text_consumer qdcxx_event_text_headers qdcxx_event_text_installed_example PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
if(ORG_TTLDTOR_QDCXX_CONSUMER_ZONE_BACKEND)
    get_filename_component(qdcxxPrefix "${qdcxxIncludeRoot}/.." ABSOLUTE)
    add_test(NAME event_text_installed_consumer COMMAND qdcxx_event_text_consumer
        "${qdcxxPrefix}/share/qdcxx/time-zone-data/tzdb-2026f")
else()
    add_test(NAME event_text_installed_consumer COMMAND qdcxx_event_text_consumer)
endif()
add_test(NAME event_text_installed_example COMMAND qdcxx_event_text_installed_example --run)
add_test(NAME event_text_installed_help COMMAND "${CMAKE_COMMAND}"
    "-DPROGRAM=$<TARGET_FILE:qdcxx_event_text_installed_example>"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/../event_text_example_help.cmake")
set_tests_properties(event_text_installed_example event_text_installed_help PROPERTIES TIMEOUT 30)
set_tests_properties(event_text_installed_consumer PROPERTIES TIMEOUT 120)
