cmake_minimum_required(VERSION 3.25)
project(qdcxx_order_consumer LANGUAGES CXX)
option(ORG_TTLDTOR_QDCXX_ORDER_CONSUMER_USE_FEED "Consume the installed Feed target and typed API" ON)
if(ORG_TTLDTOR_QDCXX_ORDER_CONSUMER_USE_FEED)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
else()
    find_package(qdcxx CONFIG REQUIRED)
endif()
add_executable(qdcxx_order_consumer main.cpp order_header.cpp)
if(ORG_TTLDTOR_QDCXX_ORDER_CONSUMER_USE_FEED)
    target_link_libraries(qdcxx_order_consumer PRIVATE qdcxx::feed)
    target_compile_definitions(qdcxx_order_consumer PRIVATE ORG_TTLDTOR_QDCXX_ORDER_CONSUMER_USE_FEED=1)
else()
    target_link_libraries(qdcxx_order_consumer PRIVATE qdcxx_order)
    target_compile_definitions(qdcxx_order_consumer PRIVATE ORG_TTLDTOR_QDCXX_ORDER_CONSUMER_USE_FEED=0)
endif()
set_target_properties(qdcxx_order_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME order_consumer COMMAND qdcxx_order_consumer)
set_tests_properties(order_consumer PROPERTIES TIMEOUT 15)
