cmake_minimum_required(VERSION 3.25)
project(qdcxx_candle_attributes_consumer LANGUAGES CXX)
option(ORG_TTLDTOR_QDCXX_CONSUMER_FEED "Consume only the installed feed target" OFF)
if(ORG_TTLDTOR_QDCXX_CONSUMER_FEED)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
else()
    find_package(qdcxx CONFIG REQUIRED)
endif()
add_executable(qdcxx_candle_attributes_consumer main.cpp)
if(ORG_TTLDTOR_QDCXX_CONSUMER_FEED)
    target_link_libraries(qdcxx_candle_attributes_consumer PRIVATE qdcxx::feed)
    target_compile_definitions(qdcxx_candle_attributes_consumer PRIVATE ORG_TTLDTOR_QDCXX_CONSUMER_FEED=1)
else()
    target_link_libraries(qdcxx_candle_attributes_consumer PRIVATE qdcxx_candle_values)
endif()
set_target_properties(qdcxx_candle_attributes_consumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME candle_attributes_consumer COMMAND qdcxx_candle_attributes_consumer)
set_tests_properties(candle_attributes_consumer PROPERTIES TIMEOUT 20)
