cmake_minimum_required(VERSION 3.25)
project(qdcxx_ipf_transfer_sdk LANGUAGES NONE)
option(ORG_TTLDTOR_QDCXX_IPF_TRANSFER_DEPENDENCY "Explicitly build the pinned private IPF dependency SDK" OFF)
if(NOT ORG_TTLDTOR_QDCXX_IPF_TRANSFER_DEPENDENCY)
    message(STATUS "IPF transfer dependency OFF: no compiler or SDK discovery/acquisition")
    return()
endif()
enable_language(CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../VerifyIpfTransferSdk.cmake")
qdcxxCheckIpfTransferHost()
set(ORG_TTLDTOR_QDCXX_IPF_TRANSFER_SOURCE "" CACHE PATH "Explicit provisioned pinned Boost source profile")
if(ORG_TTLDTOR_QDCXX_IPF_TRANSFER_SOURCE STREQUAL "" OR NOT IS_DIRECTORY "${ORG_TTLDTOR_QDCXX_IPF_TRANSFER_SOURCE}")
    message(FATAL_ERROR "QDCXX_IPF_SOURCE: provide explicit ORG_TTLDTOR_QDCXX_IPF_TRANSFER_SOURCE; see tools/deps/IPF-TRANSFER.md")
endif()
set(manifest "${CMAKE_CURRENT_LIST_DIR}/boost-1.92.0.sources.json")
qdcxxVerifyIpfTransferSources("${manifest}" "${ORG_TTLDTOR_QDCXX_IPF_TRANSFER_SOURCE}")
file(READ "${manifest}" profile)
string(JSON count LENGTH "${profile}" files)
math(EXPR last "${count}-1")
set(sources)
foreach(index RANGE 0 ${last})
    string(JSON relative GET "${profile}" files ${index} path)
    if(relative MATCHES "^libs/url/src/.*\\.cpp$")
        list(APPEND sources "${ORG_TTLDTOR_QDCXX_IPF_TRANSFER_SOURCE}/${relative}")
    endif()
endforeach()
add_library(qdcxx_boost_url STATIC ${sources})
set_target_properties(qdcxx_boost_url PROPERTIES EXPORT_NAME url CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_compile_features(qdcxx_boost_url PUBLIC cxx_std_23)
target_include_directories(qdcxx_boost_url SYSTEM PUBLIC
    "$<BUILD_INTERFACE:${ORG_TTLDTOR_QDCXX_IPF_TRANSFER_SOURCE}>" "$<INSTALL_INTERFACE:source>")
target_compile_definitions(qdcxx_boost_url PUBLIC BOOST_ALL_NO_LIB=1 BOOST_URL_NO_LIB=1 BOOST_URL_STATIC_LINK=1 BOOST_BEAST_HEADER_ONLY=1
    PRIVATE BOOST_URL_SOURCE=1)
if(WIN32)
    target_compile_definitions(qdcxx_boost_url PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
    set(platform "windows-x64-msvc1942-release-md")
    set(archive "lib/qdcxx_boost_url.lib")
else()
    set(platform "linux-x64-gcc13.3-debug-asan-ubsan")
    set(archive "lib/libqdcxx_boost_url.a")
endif()
add_executable(qdcxx_ipf_transfer_probe "${CMAKE_CURRENT_LIST_DIR}/../../tests/conformance/ipf-transfer-dependency/parser_probe.cpp")
target_link_libraries(qdcxx_ipf_transfer_probe PRIVATE qdcxx_boost_url)
enable_testing()
add_test(NAME qdcxx_ipf_transfer_dependency_probe COMMAND qdcxx_ipf_transfer_probe)
set_tests_properties(qdcxx_ipf_transfer_dependency_probe PROPERTIES TIMEOUT 30)
file(SHA256 "${manifest}" sourceManifestHash)
# Artifact identity is per build. This metadata contains no source/build absolute paths.
foreach(value IN ITEMS CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE CMAKE_MSVC_RUNTIME_LIBRARY)
    if("${${value}}" MATCHES "[\"\\\\\r\n;]")
        message(FATAL_ERROR "QDCXX_IPF_HOST: unsupported metadata separator in ${value}")
    endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build-profile.json"
    "{\"schemaVersion\":1,\"platform\":\"${platform}\",\"compilerId\":\"${CMAKE_CXX_COMPILER_ID}\",\"compilerVersion\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"buildType\":\"${CMAKE_BUILD_TYPE}\",\"crt\":\"${CMAKE_MSVC_RUNTIME_LIBRARY}\",\"flags\":\"${CMAKE_CXX_FLAGS}\",\"debugFlags\":\"${CMAKE_CXX_FLAGS_DEBUG}\",\"releaseFlags\":\"${CMAKE_CXX_FLAGS_RELEASE}\",\"archive\":\"${archive}\",\"sourceManifestSha256\":\"${sourceManifestHash}\",\"macros\":\"BOOST_ALL_NO_LIB=1 BOOST_URL_NO_LIB=1 BOOST_URL_STATIC_LINK=1 BOOST_BEAST_HEADER_ONLY=1; Windows _WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN\"}\n")
configure_file(SdkConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/qdcxx_ipf_transfer_sdkConfig.cmake" @ONLY)
install(TARGETS qdcxx_boost_url EXPORT qdcxxIpfTransferSdkTargets ARCHIVE DESTINATION lib)
install(EXPORT qdcxxIpfTransferSdkTargets NAMESPACE qdcxx_private:: DESTINATION lib/cmake/qdcxx_ipf_transfer_sdk)
install(DIRECTORY "${ORG_TTLDTOR_QDCXX_IPF_TRANSFER_SOURCE}/" DESTINATION source)
install(FILES "${manifest}" "${CMAKE_CURRENT_BINARY_DIR}/build-profile.json" DESTINATION share/qdcxx-ipf-transfer)
install(FILES "${CMAKE_CURRENT_LIST_DIR}/../VerifyIpfTransferSdk.cmake" DESTINATION lib/cmake/qdcxx_ipf_transfer_sdk)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/qdcxx_ipf_transfer_sdkConfig.cmake" DESTINATION lib/cmake/qdcxx_ipf_transfer_sdk)
