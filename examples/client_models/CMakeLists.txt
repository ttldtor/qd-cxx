cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(qdcxx_client_models_example LANGUAGES CXX)
endif()
if(NOT TARGET qdcxx_feed OR NOT TARGET qdcxx_ipf)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
endif()
if(NOT TARGET qdcxx_feed OR NOT TARGET qdcxx_ipf)
    message(FATAL_ERROR "The installed package requires ORG_TTLDTOR_QDCXX_BUILD_FEED_API=ON and ORG_TTLDTOR_QDCXX_BUILD_IPF=ON")
endif()
add_executable(qdcxx_client_models main.cpp)
target_link_libraries(qdcxx_client_models PRIVATE qdcxx_feed qdcxx_ipf)
target_compile_options(qdcxx_client_models PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
set_target_properties(qdcxx_client_models PROPERTIES CXX_EXTENSIONS OFF OUTPUT_NAME qdcxx-client-models)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    include(CTest)
    if(BUILD_TESTING)
        add_test(NAME client_models_help COMMAND "${CMAKE_COMMAND}" "-DPROGRAM=$<TARGET_FILE:qdcxx_client_models>" -P "${CMAKE_CURRENT_SOURCE_DIR}/check.cmake")
        add_test(NAME client_models_values COMMAND qdcxx_client_models --run)
        set_tests_properties(client_models_values PROPERTIES PASS_REGULAR_EXPRESSION "candles=1 bids=1 profiles=1 option-chains=0")
    endif()
endif()
