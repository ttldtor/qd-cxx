cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(qdcxx_client_queries_example LANGUAGES CXX)
endif()
if(NOT TARGET qdcxx_feed)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
endif()
if(NOT TARGET qdcxx_feed)
    message(FATAL_ERROR "The installed package requires ORG_TTLDTOR_QDCXX_BUILD_FEED_API=ON")
endif()
add_executable(qdcxx_client_queries main.cpp)
target_link_libraries(qdcxx_client_queries PRIVATE qdcxx_feed)
target_compile_options(qdcxx_client_queries PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
set_target_properties(qdcxx_client_queries PROPERTIES CXX_EXTENSIONS OFF OUTPUT_NAME qdcxx-client-queries)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    include(CTest)
    if(BUILD_TESTING)
        add_test(NAME client_queries_help COMMAND "${CMAKE_COMMAND}" "-DPROGRAM=$<TARGET_FILE:qdcxx_client_queries>" -P "${CMAKE_CURRENT_SOURCE_DIR}/check.cmake")
    endif()
endif()
