cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(qdcxx_feed_market_example LANGUAGES CXX)
endif()
if(NOT TARGET qdcxx_feed)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
endif()
if(NOT TARGET qdcxx_feed)
    message(FATAL_ERROR "The installed package requires BUILD_FEED_API=ON")
endif()
find_package(Threads REQUIRED)
add_executable(qdcxx_feed_market main.cpp signal_stop.cpp)
target_link_libraries(qdcxx_feed_market PRIVATE qdcxx_feed Threads::Threads)
target_compile_options(qdcxx_feed_market PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
set_target_properties(qdcxx_feed_market PROPERTIES CXX_EXTENSIONS OFF OUTPUT_NAME qdcxx-feed-market)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    include(CTest)
    if(BUILD_TESTING)
        add_test(NAME feed_market_help COMMAND "${CMAKE_COMMAND}" "-DPROGRAM=$<TARGET_FILE:qdcxx_feed_market>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/help_test.cmake")
    endif()
endif()
