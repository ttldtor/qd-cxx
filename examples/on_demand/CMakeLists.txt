cmake_minimum_required(VERSION 3.25)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(qdcxx_on_demand_example LANGUAGES CXX)
    include(CTest)
endif()
if(NOT TARGET qdcxx::feed)
    find_package(qdcxx CONFIG REQUIRED COMPONENTS feed)
endif()
find_package(Threads REQUIRED)

# Installed examples carry the same synthetic peer and fixture used by conformance tests.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/conformance/recovery_test_peer.cpp")
    set(peer_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/conformance")
    set(fixture_dir "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/ondemand")
else()
    set(peer_dir "${CMAKE_CURRENT_SOURCE_DIR}")
    set(fixture_dir "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
add_executable(qdcxx_on_demand_example main.cpp fixture_peer.cpp "${peer_dir}/recovery_test_peer.cpp")
target_include_directories(qdcxx_on_demand_example PRIVATE "${peer_dir}" "${fixture_dir}")
target_link_libraries(qdcxx_on_demand_example PRIVATE qdcxx::feed Threads::Threads)
target_compile_options(qdcxx_on_demand_example PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
if(WIN32)
    target_compile_definitions(qdcxx_on_demand_example PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(qdcxx_on_demand_example PRIVATE ws2_32)
endif()
set_target_properties(qdcxx_on_demand_example PROPERTIES CXX_EXTENSIONS OFF OUTPUT_NAME qdcxx-on-demand)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR AND BUILD_TESTING)
    add_test(NAME on_demand_example COMMAND "${CMAKE_COMMAND}" "-DPROGRAM=$<TARGET_FILE:qdcxx_on_demand_example>"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/check.cmake")
    set_tests_properties(on_demand_example PROPERTIES TIMEOUT 20)
endif()
