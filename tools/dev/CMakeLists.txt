target_include_directories(qdcxx_logger_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(qdcxx_logger_tests PROPERTIES CXX_EXTENSIONS OFF)

# Mutate private generated copies only; production launcher sources stay unchanged.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/run_utc.cpp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/run_utc.cpp" logger_source)
foreach(kind args decoding exit fault)
    if(kind STREQUAL "args")
        string(REPLACE "const std::vector<std::string> command(arguments.begin() + 3, arguments.end());"
            "std::vector<std::string> command(arguments.begin() + 3, arguments.end()); command.push_back(\"mutation\");"
            variant "${logger_source}")
    elseif(kind STREQUAL "decoding")
        string(REPLACE "append(REPLACEMENT, emit);" "append(\"?\", emit);" variant "${logger_source}")
    elseif(kind STREQUAL "exit")
        string(REPLACE "return result;" "return result == 37 ? 0 : result;" variant "${logger_source}")
    else()
        string(REPLACE "log.write(line);"
            "log.write(line); if (message == \"orphan-ready\") { std::this_thread::sleep_for(std::chrono::milliseconds(100)); throw std::runtime_error(\"injected destination flush failure\"); }"
            variant "${logger_source}")
        string(PREPEND variant "#include <thread>\n")
    endif()
    if(variant STREQUAL logger_source)
        message(FATAL_ERROR "Logger ${kind} mutation no longer matches its source")
    endif()
    set(variant_source "${CMAKE_CURRENT_BINARY_DIR}/generated/${kind}.cpp")
    file(CONFIGURE OUTPUT "${variant_source}" CONTENT "${variant}" @ONLY)
    if(kind STREQUAL "fault")
        set(target qdcxx_run_utc_fault)
    else()
        set(target "qdcxx_run_utc_mutant_${kind}")
    endif()
    add_executable(${target} "${variant_source}" "${process_source}")
    target_compile_features(${target} PRIVATE cxx_std_23)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
endforeach()
